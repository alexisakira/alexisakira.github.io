<section class="research-area" id="asset-price-bubbles" aria-labelledby="asset-price-bubbles-heading">
  <header class="research-area__header">
    <p class="research-area__eyebrow">Research area</p>
    <h2 id="asset-price-bubbles-heading">Asset price bubbles</h2>

<section class="research-area" id="power-laws" aria-labelledby="power-laws-heading">
  <header class="research-area__header">
    <p class="research-area__eyebrow">Research area</p>

    <li>

    </li>
    <li>
      <a href="https://doi.org/10.1016/j.physd.2020.132649">On the Emergence of a Power Law in the Distribution of COVID-19 Cases</a>

<section class="research-area" id="macro-finance" aria-labelledby="macro-finance-heading">

    <p class="research-area__eyebrow">Research area</p>
